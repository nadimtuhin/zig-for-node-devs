Appendix 06 • Cloud & Containers

Production Docker Containerization

Shrinking 300MB Node.js containers into a pristine 4MB scratch image with zero CVE vulnerabilities.

# Multi-stage Dockerfile for Zig
FROM alpine:3.20 AS builder
RUN apk add --no-cache zig
WORKDIR /app
COPY . .
RUN zig build -Doptimize=ReleaseSafe -Dtarget=x86_64-linux-musl

# Final production stage
FROM scratch
COPY --from=builder /app/zig-out/bin/my_app /server
EXPOSE 8080
ENTRYPOINT ["/server"]
