LeetCode in Idiomatic Zig

Classic algorithmic patterns solved with zero-allocation stack buffers and cache-friendly data structures.

1. Two Sum — O(N) Hash Table Lookup

Node.js uses new Map() which allocates individual bucket nodes on the V8 heap. Zig's AutoHashMap uses a flat Robin Hood / Swiss-table layout with contiguous memory.

const std = @import("std");

fn twoSum(allocator: std.mem.Allocator, nums: []const i32, target: i32) !?[2]usize {
    var map = std.AutoHashMap(i32, usize).init(allocator);
    defer map.deinit();

    for (nums, 0..) |num, idx| {
        const complement = target - num;
        if (map.get(complement)) |prev_idx| {
            return .{ prev_idx, idx };
        }
        try map.put(num, idx);
    }
    return null;
}

2. Valid Parentheses — Zero-Allocation Fixed Stack

Rather than allocating dynamic arrays, Zig solves parsing problems with a pre-sized stack buffer and single-pass byte matching.

fn isValid(s: []const u8) bool {
    var stack: [256]u8 = undefined;
    var top: usize = 0;

    for (s) |char| {
        switch (char) {
            '(', '{', '[' => {
                if (top >= stack.len) return false;
                stack[top] = char;
                top += 1;
            },
            ')' => { if (top == 0 or stack[top - 1] != '(') return false; top -= 1; },
            '}' => { if (top == 0 or stack[top - 1] != '{') return false; top -= 1; },
            ']' => { if (top == 0 or stack[top - 1] != '[') return false; top -= 1; },
            else => {},
        }
    }
    return top == 0;
}

3. Reverse Linked List — In-Place Pointer Manipulation

Node.js manipulates JavaScript object references with V8 garbage collection overhead. Zig reverses pointers in place with zero heap allocations.

const Node = struct {
    val: i32,
    next: ?*Node = null,
};

fn reverseList(head: ?*Node) ?*Node {
    var prev: ?*Node = null;
    var current = head;

    while (current) |curr| {
        const next_node = curr.next;
        curr.next = prev;
        prev = curr;
        current = next_node;
    }
    return prev;
}

4. Binary Search — Overflow-Free Slicing

Node.js requires Math.floor((left + right) / 2) with 64-bit float math. Zig performs integer arithmetic and bounds slicing with zero overhead.

fn binarySearch(nums: []const i32, target: i32) ?usize {
    if (nums.len == 0) return null;
    var left: usize = 0;
    var right: usize = nums.len;

    while (left < right) {
        const mid = left + (right - left) / 2;
        if (nums[mid] == target) {
            return mid;
        } else if (nums[mid] < target) {
            left = mid + 1;
        } else {
            right = mid;
        }
    }
    return null;
}
