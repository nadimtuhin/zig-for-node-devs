Appendix 03 • Native Integration

C Interop & True Node-API Addons

Replacing `node-gyp`, Python dependencies, and complex C++ binding boilerplate with pure Zig and zero external build tools.

🔌 Zig is a First-Class C Compiler

Building native C++ addons for Node.js typically requires Python, CMake/Make, Visual Studio C++ build tools, and node-gyp. A single OS mismatch breaks npm installs.

Zig embeds a full Clang C/C++ compiler and can directly parse C headers via @cImport without generating wrapper code or running pre-build steps.

1. Direct C Header Ingestion with `@cImport`

Import C headers directly into Zig. Zig handles C types, pointers, and struct layouts seamlessly:

c_interop.zig Zig 
const std = @import("std");

// Import standard C headers directly
const c = @cImport({
    @cInclude("stdio.h");
    @cInclude("math.h");
});

pub fn main() void {
    // 1. Call standard C math functions
    const result = c.sqrt(144.0);
    std.debug.print("Square root from C math.h: {d}\n", .{result}); // 12.0

    // 2. Interacting with C strings: converting [*:0]const u8 to Zig slice []const u8
    const c_str: [*:0]const u8 = "Hello from C string";
    const zig_slice: []const u8 = std.mem.span(c_str);
    std.debug.print("Zig slice length: {d}, content: {s}\n", .{ zig_slice.len, zig_slice });
}

2. Building a True Native Node-API (.node) Addon

You can build a native .node binary that works with require('./addon.node') in Node.js using Node-API C bindings:

addon.zig (Native Node-API Addon) Zig 
const std = @import("std");

// Import Node-API C declarations
const napi = @cImport({
    @cInclude("node_api.h");
});

// Implementation of the exported function: fastMultiply(a, b)
fn fastMultiply(env: napi.napi_env, info: napi.napi_callback_info) callconv(.C) napi.napi_value {
    var argc: usize = 2;
    var argv: [2]napi.napi_value = undefined;
    _ = napi.napi_get_cb_info(env, info, &argc, &argv, null, null);

    var a: f64 = 0;
    var b: f64 = 0;
    _ = napi.napi_get_value_double(env, argv[0], &a);
    _ = napi.napi_get_value_double(env, argv[1], &b);

    var result: napi.napi_value = undefined;
    _ = napi.napi_create_double(env, a * b, &result);
    return result;
}

// Module initialization callback
export fn napi_register_module_v1(env: napi.napi_env, exports: napi.napi_value) napi.napi_value {
    var fn_val: napi.napi_value = undefined;
    _ = napi.napi_create_function(
        env,
        "fastMultiply",
        napi.NAPI_AUTO_LENGTH,
        fastMultiply,
        null,
        &fn_val,
    );
    _ = napi.napi_set_named_property(env, exports, "fastMultiply", fn_val);
    return exports;
}

3. Compile & Require in Node.js

Compile the shared library using Zig with Node.js include headers, then load directly:

# Single command build for native Node addon (zero node-gyp needed!) zig build-lib addon.zig -dynamic -lc \ -I $(node -p "path.dirname(process.execPath)")/include/node \ -O ReleaseFast -femit-bin=addon.node
index.js (Consuming in Node.js) JavaScript 
import { createRequire } from 'module';
const require = createRequire(import.meta.url);

// Load the compiled Zig native addon directly
const addon = require('./addon.node');

console.log(addon.fastMultiply(6.5, 4.0)); // 26.0 (running at bare-metal native CPU speed!)

⚡ Alternative: Zero-Header Bun FFI

If using Bun, you don't even need Node-API headers. Export a raw C-ABI function from Zig and consume it via bun:ffi:

import { dlopen, FFIType } from "bun:ffi";

const { symbols: { fastMultiply } } = dlopen("./libaddon.so", {
  fastMultiply: {
    args: [FFIType.f64, FFIType.f64],
    returns: FFIType.f64,
  },
});

console.log(fastMultiply(10.5, 2.0)); // 21.0
